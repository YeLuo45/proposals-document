## 1. 背景

当前MultiAgentPanel通过轮询sessionStorage获取任务状态，但OrchestratorAgent的任务状态只存在于内存中，从未写入sessionStorage。

**目标：** 让OrchestratorAgent在任务状态变化时自动同步到sessionStorage，MultiAgentPanel能实时显示。

---

## 2. 功能设计

### 2.1 创建MultiAgentStore

```typescript
// src/services/agents/multiAgentStore.ts

import type { Task } from './types'

const STORAGE_KEY = 'multiAgentTasks'

interface StoreData {
  traceId: string | null
  tasks: Task[]
  lastUpdated: number
}

class MultiAgentStore {
  save(data: { traceId: string; tasks: Task[] }): void {
    const storeData: StoreData = {
      traceId: data.traceId,
      tasks: data.tasks,
      lastUpdated: Date.now(),
    }
    try {
      sessionStorage.setItem(STORAGE_KEY, JSON.stringify(storeData))
    } catch (e) {
      console.warn('[MultiAgentStore] save failed:', e)
    }
  }

  load(): StoreData | null {
    try {
      const raw = sessionStorage.getItem(STORAGE_KEY)
      if (!raw) return null
      return JSON.parse(raw) as StoreData
    } catch (e) {
      console.warn('[MultiAgentStore] load failed:', e)
      return null
    }
  }

  clear(): void {
    try {
      sessionStorage.removeItem(STORAGE_KEY)
    } catch (e) {
      console.warn('[MultiAgentStore] clear failed:', e)
    }
  }

  getTasks(): Task[] {
    const data = this.load()
    return data?.tasks || []
  }

  getTraceId(): string | null {
    const data = this.load()
    return data?.traceId || null
  }
}

export const multiAgentStore = new MultiAgentStore()
```

### 2.2 修改OrchestratorAgent

```typescript
// 修改 src/services/agents/orchestratorAgent.ts

import { agentBus, agentRegistry } from './agentRegistry'
import type { AgentMessage, Task, AgentConfig } from './types'
import { AgentType } from './types'
import { multiAgentStore } from './multiAgentStore'  // 新增

let taskCounter = 0

export class OrchestratorAgent {
  private id = 'orchestrator'
  private pendingTasks = new Map<string, Task>()
  private currentTraceId: string | null = null  // 新增

  constructor() {
    agentBus.subscribe(this.id, this.handleMessage.bind(this))
  }

  // ... 其他现有方法 ...

  private async handleTaskUpdate(): Promise<void> {
    // 新增：每次任务状态变化时同步到sessionStorage
    const tasks = Array.from(this.pendingTasks.values())
    if (this.currentTraceId && tasks.length > 0) {
      multiAgentStore.save({
        traceId: this.currentTraceId,
        tasks,
      })
    }
  }

  async processUserRequest(input: { message: string }): Promise<void> {
    const parentId = `task-${++taskCounter}`
    this.currentTraceId = parentId  // 新增：设置当前traceId

    const subTasks = this.decomposeTask(parentId, input.message)
    subTasks.forEach(t => this.pendingTasks.set(t.id, t))

    // 同步到sessionStorage
    multiAgentStore.save({
      traceId: parentId,
      tasks: Array.from(this.pendingTasks.values()),
    })

    for (const task of subTasks) {
      const executor = this.selectExecutor(task)
      task.assignedTo = executor.id

      await agentBus.send({
        from: this.id,
        to: executor.id,
        type: 'task',
        payload: task,
        timestamp: Date.now(),
        traceId: parentId,
      })
    }

    console.log(`[Orchestrator] Started ${subTasks.length} tasks`)
  }

  // 修改 handleResult
  private async handleResult(payload: { taskId: string; result: unknown }): Promise<void> {
    const task = this.pendingTasks.get(payload.taskId)
    if (!task) return

    task.status = 'completed'
    task.result = payload.result
    task.completedAt = Date.now()
    this.pendingTasks.delete(payload.taskId)

    // 同步到sessionStorage
    this.handleTaskUpdate()

    if (this.pendingTasks.size === 0) {
      console.log('[Orchestrator] All tasks completed')
      // 3秒后清理
      setTimeout(() => multiAgentStore.clear(), 3000)
    }
  }

  // 修改 handleError
  private async handleError(payload: { taskId: string; error: string }): Promise<void> {
    const task = this.pendingTasks.get(payload.taskId)
    if (task) {
      task.status = 'failed'
      this.pendingTasks.delete(payload.taskId)
    }

    // 同步到sessionStorage
    this.handleTaskUpdate()
  }
}
```

// 在任务开始时同步状态
private async executeTask(task: Task, traceId?: string): Promise<void> {
  try {
    task.status = 'in_progress'

    // 同步到sessionStorage（获取当前pending tasks并更新）
    this.syncTaskStatus(task)

    // ... 原有代码 ...
  }
}

private syncTaskStatus(task: Task): void {
  // 从multiAgentStore获取当前状态，追加当前任务
  const current = multiAgentStore.load()
  if (current) {
    const existingIndex = current.tasks.findIndex(t => t.id === task.id)
    if (existingIndex >= 0) {
      current.tasks[existingIndex] = task
    } else {
      current.tasks.push(task)
    }
    multiAgentStore.save(current)
  }
}

## 3. 数据流程

```
OrchestratorAgent.processUserRequest()
    ↓ 创建任务
    ↓ multiAgentStore.save(tasks)
    ↓
    ├─→ ExecutorAgent.executeTask()
    │       ↓ 更新任务状态
    │       ↓ multiAgentStore.save(tasks)
    │       ↓
    └─→ MultiAgentPanel (每1秒轮询)
            ↓ sessionStorage.getItem
            ↓ 显示最新状态
```
