## 1. 背景与目标

V67实现的记忆上下文（MemoryManager）只在当前会话有效，刷新页面后丢失。

**目标：** 将记忆持久化到IndexedDB，实现跨会话记忆复用。

---

## 2. 现状分析

### 2.1 V67记忆模块
- `memoryManager.ts` — 单例，内存中存储 `MemoryEntry[]`
- `messageHistoryMemory.ts` — 从MessageHistory提取记忆
- 无持久化：刷新页面 → 记忆清空

### 2.2 现有存储
- IndexedDB：`messageHistory` 表（已存在）
- 需要新增：`agentMemory` 表

## 3. 功能设计

### 3.1 新增MemoryStorage服务
```typescript
// src/services/storage/memoryStorage.ts

import type { MemoryEntry } from '../agent/memory/MemoryContext'

const DB_NAME = 'pixelpal'
const STORE_NAME = 'agentMemory'
const DB_VERSION = 1

export async function initMemoryStore(): Promise<IDBDatabase> {
  return new Promise((resolve, reject) => {
    const request = indexedDB.open(DB_NAME, DB_VERSION)
    request.onerror = () => reject(request.error)
    request.onsuccess = () => resolve(request.result)
    request.onupgradeneeded = (event) => {
      const db = (event.target as IDBOpenDBRequest).result
      if (!db.objectStoreNames.contains(STORE_NAME)) {
        const store = db.createObjectStore(STORE_NAME, { keyPath: 'id' })
        store.createIndex('type', 'type', { unique: false })
        store.createIndex('importance', 'importance', { unique: false })
        store.createIndex('createdAt', 'createdAt', { unique: false })
      }
    }
  })
}

export async function saveMemoryEntry(entry: MemoryEntry): Promise<void> {
  const db = await initMemoryStore()
  return new Promise((resolve, reject) => {
    const tx = db.transaction(STORE_NAME, 'readwrite')
    tx.objectStore(STORE_NAME).put(entry)
    tx.oncomplete = () => resolve()
    tx.onerror = () => reject(tx.error)
  })
}

export async function loadMemoryEntries(limit = 100): Promise<MemoryEntry[]> {
  const db = await initMemoryStore()
  return new Promise((resolve, reject) => {
    const tx = db.transaction(STORE_NAME, 'readonly')
    const store = tx.objectStore(STORE_NAME)
    const index = store.index('createdAt')
    const results: MemoryEntry[] = []
    const request = index.openCursor(null, 'prev')  // 最新在前
    request.onsuccess = () => {
      const cursor = request.result
      if (cursor && results.length < limit) {
        results.push(cursor.value)
        cursor.continue()
      } else {
        resolve(results)
      }
    }
    request.onerror = () => reject(request.error)
  })
}

export async function clearMemoryEntries(): Promise<void> {
  const db = await initMemoryStore()
  return new Promise((resolve, reject) => {
    const tx = db.transaction(STORE_NAME, 'readwrite')
    tx.objectStore(STORE_NAME).clear()
    tx.oncomplete = () => resolve()
    tx.onerror = () => reject(tx.error)
  })
}

export async function deleteMemoryEntry(id: string): Promise<void> {
  const db = await initMemoryStore()
  return new Promise((resolve, reject) => {
    const tx = db.transaction(STORE_NAME, 'readwrite')
    tx.objectStore(STORE_NAME).delete(id)
    tx.oncomplete = () => resolve()
    tx.onerror = () => reject(tx.error)
  })
}
```

### 3.2 扩展MemoryManager

修改 `src/services/agent/memory/memoryManager.ts`：

```typescript
// 新增持久化相关方法
import { saveMemoryEntry, loadMemoryEntries, clearMemoryEntries } from '../../storage/memoryStorage'

class MemoryManager {
  private persistenceEnabled = false

  async enablePersistence(): Promise<void> {
    await initMemoryStore()
    // 加载已有记忆到内存
    const entries = await loadMemoryEntries()
    for (const entry of entries) {
      this.context.entries.set(entry.id, entry)
    }
    this.persistenceEnabled = true
  }

  async add(entry: Omit<MemoryEntry, 'id' | 'createdAt'>): Promise<void> {
    const fullEntry: MemoryEntry = {
      ...entry,
      id: crypto.randomUUID(),
      createdAt: Date.now(),
    }
    this.context.entries.set(fullEntry.id, fullEntry)
    if (this.persistenceEnabled) {
      await saveMemoryEntry(fullEntry)
    }
  }

  async clear(): Promise<void> {
    super.clear()  // 清理内存
    if (this.persistenceEnabled) {
      await clearMemoryEntries()
    }
  }

  // 获取记忆统计
  async getStats(): Promise<{ count: number; byType: Record<string, number> }> {
    const entries = this.persistenceEnabled
      ? await loadMemoryEntries()
      : Array.from(this.context.entries.values())
    return {
      count: entries.length,
      byType: entries.reduce((acc, e) => {
        acc[e.type] = (acc[e.type] || 0) + 1
        return acc
      }, {} as Record<string, number>),
    }
  }
}
```

### 3.3 记忆过期清理

```typescript
// memoryStorage.ts 新增
export async function cleanExpiredEntries(): Promise<void> {
  const db = await initMemoryStore()
  return new Promise((resolve, reject) => {
    const tx = db.transaction(STORE_NAME, 'readwrite')
    const store = tx.objectStore(STORE_NAME)
    const now = Date.now()
    const request = store.openCursor()
    request.onsuccess = () => {
      const cursor = request.result
      if (cursor) {
        if (cursor.value.expiresAt && cursor.value.expiresAt < now) {
          cursor.delete()
        }
        cursor.continue()
      } else {
        resolve()
      }
    }
    request.onerror = () => reject(request.error)
  })
}
```

### 3.4 初始化时机

在 `App.tsx` 或入口组件中：

useEffect(() => {
  memoryManager.enablePersistence()
}, [])

## 4. 非功能需求

- 记忆持久化是**可选项**，默认关闭（向后兼容）
- 调用 `enablePersistence()` 后才启用持久化
- 过期条目自动清理
- 不影响现有MemoryManager API
