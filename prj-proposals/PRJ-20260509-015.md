## 1. 背景与目标

当前PixelPal是单一Agent（AgentExecutor）处理所有任务。在V64-V76迭代中，我们构建了完整的基础设施：
- V64: AgentExecutor核心循环
- V67: 记忆上下文注入
- V73: 情感上下文注入
- V74: 场景上下文注入
- V75: UI层信号采集

**下一阶段目标：** 从单一Agent扩展为多Agent协作系统，让不同专业Agent协同处理复杂任务。

```typescript
  completedAt?: number
```

### 2.2 AgentRegistry

```typescript
// src/services/agents/agentRegistry.ts

import { AgentType, type AgentConfig } from './types'

class AgentRegistry {
  private agents = new Map<string, AgentConfig>()

  register(config: AgentConfig): void {
    this.agents.set(config.id, config)
  }

  get(id: string): AgentConfig | undefined {
    return this.agents.get(id)
  }

  getByType(type: AgentType): AgentConfig[] {
    return Array.from(this.agents.values()).filter(a => a.type === type)
  }

  list(): AgentConfig[] {
    return Array.from(this.agents.values())
  }

  unregister(id: string): void {
    this.agents.delete(id)
  }
}

export const agentRegistry = new AgentRegistry()
```

### 2.3 AgentBus（消息总线）

```typescript
// src/services/agents/agentBus.ts

import type { AgentMessage } from './types'

type MessageHandler = (msg: AgentMessage) => void | Promise<void>

class AgentBus {
  private handlers = new Map<string, Set<MessageHandler>>()
  private agentQueues = new Map<string, AgentMessage[]>()
  private processing = new Set<string>()

  // 订阅消息
  subscribe(agentId: string, handler: MessageHandler): () => void {
    if (!this.handlers.has(agentId)) {
      this.handlers.set(agentId, new Set())
    }
    this.handlers.get(agentId)!.add(handler)

    // 返回取消订阅函数
    return () => {
      this.handlers.get(agentId)?.delete(handler)
    }
  }

  // 发送消息
  async send(msg: AgentMessage): Promise<void> {
    // 放入接收者队列
    if (!this.agentQueues.has(msg.to)) {
      this.agentQueues.set(msg.to, [])
    }
    this.agentQueues.get(msg.to)!.push(msg)

    // 异步处理
    this.processQueue(msg.to)
  }

  // 广播消息
  async broadcast(msg: Omit<AgentMessage, 'to'>): Promise<void> {
    const allAgents = Array.from(this.handlers.keys())
    await Promise.all(
      allAgents
        .filter(id => id !== msg.from)  // 不发给自己
        .map(id => this.send({ ...msg, to: id }))
    )
  }

  // 处理队列
  private async processQueue(agentId: string): Promise<void> {
    if (this.processing.has(agentId)) return
    this.processing.add(agentId)

    try {
      while (this.agentQueues.has(agentId) && this.agentQueues.get(agentId)!.length > 0) {
        const msg = this.agentQueues.get(agentId)!.shift()!
        const handlers = this.handlers.get(agentId)
        if (handlers) {
          await Promise.all(
            Array.from(handlers).map(h => {
              try {
                const result = h(msg)
                if (result instanceof Promise) return result.catch(console.error)
              } catch (e) {
                console.error(`[AgentBus] Handler error for ${agentId}:`, e)
              }
            })
          )
        }
      }
    } finally {
      this.processing.delete(agentId)
    }
  }
}

export const agentBus = new AgentBus()
```

### 2.4 OrchestratorAgent

```typescript
// src/services/agents/orchestratorAgent.ts

import { agentBus, agentRegistry } from './agentRegistry'
import type { AgentMessage, Task } from './types'
import { v4 as uuid } from 'uuid'

class OrchestratorAgent {
  private id = 'orchestrator'

  async processUserRequest(userMessage: string): Promise<Task[]> {
    // 1. 分析任务，决定需要哪些Agent协作
    const subTasks = await this.decomposeTask(userMessage)

    // 2. 分配任务给合适的Agent
    for (const task of subTasks) {
      const assignedAgent = this.selectAgent(task)
      task.assignedTo = assignedAgent

      // 发送任务消息
      await agentBus.send({
        from: this.id,
        to: assignedAgent,
        type: 'task',
        payload: task,
        timestamp: Date.now(),
        traceId: uuid(),
      })
    }

    return subTasks
  }

  private async decomposeTask(userMessage: string): Promise<Task[]> {
    // 调用LLM分解任务（复用V64的decomposeGoal逻辑）
    // 这里简化处理，实际应该调用LLM
    return [{
      id: uuid(),
      type: 'single',
      description: userMessage,
      inputs: { message: userMessage },
      status: 'pending',
      createdAt: Date.now(),
    }]
  }

  private selectAgent(task: Task): string {
    // 根据任务类型选择Agent
    const agents = agentRegistry.getByType('executor')
    return agents[0]?.id || 'executor'
  }
}

export const orchestratorAgent = new OrchestratorAgent()
```

### 2.5 ExecutorAgent

```typescript
// src/services/agents/executorAgent.ts

import { agentBus, agentRegistry } from './agentRegistry'
import type { AgentMessage, Task } from './types'

class ExecutorAgent {
  private id = 'executor'

  constructor() {
    // 订阅AgentBus消息
    agentBus.subscribe(this.id, this.handleMessage.bind(this))
  }

  private async handleMessage(msg: AgentMessage): Promise<void> {
    if (msg.type === 'task') {
      const task = msg.payload as Task
      await this.executeTask(task)
    }
  }

  async executeTask(task: Task): Promise<void> {
    try {
      // 实际执行任务（复用V64的executeTask逻辑）
      const result = { output: `Executed: ${task.description}` }

      // 发送结果给编排Agent
      await agentBus.send({
        from: this.id,
        to: 'orchestrator',
        type: 'result',
        payload: { taskId: task.id, result },
        timestamp: Date.now(),
        traceId: msg.traceId,
      })
    } catch (error) {
      await agentBus.send({
        from: this.id,
        to: 'orchestrator',
        type: 'error',
        payload: { taskId: task.id, error: String(error) },
        timestamp: Date.now(),
        traceId: msg.traceId,
      })
    }
  }
}

export const executorAgent = new ExecutorAgent()
```
