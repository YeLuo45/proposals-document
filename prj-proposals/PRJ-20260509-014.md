## 2. Telegram实现

### 2.1 Bot Token配置

在 `.env` 中配置：


### 2.2 Long Polling实现

```typescript
// src/platform/TelegramAdapter.ts 完整实现

import type { PlatformAdapter, PlatformMessage } from './PlatformAdapter'

const API_BASE = 'https://api.telegram.org/bot'

export class TelegramAdapter implements PlatformAdapter {
  platform = 'telegram' as const
  private token: string
  private offset = 0
  private handlers: ((msg: PlatformMessage) => void)[] = []
  private polling = false
  private pollInterval: ReturnType<typeof setInterval> | null = null

  constructor() {
    this.token = import.meta.env.VITE_TELEGRAM_BOT_TOKEN || ''
  }

  private get api(): string {
    return `${API_BASE}${this.token}`
  }

  private async getUpdates(): Promise<any[]> {
    if (!this.token) return []
    try {
      const resp = await fetch(`${this.api}/getUpdates?offset=${this.offset}&timeout=10`)
      const data = await resp.json()
      return data.ok ? data.result : []
    } catch (e) {
      console.error('[Telegram] getUpdates error:', e)
      return []
    }
  }

  async sendMessage(text: string): Promise<void> {
    if (!this.token) {
      console.log('[Telegram] Bot token not configured')
      return
    }
    try {
      await fetch(`${this.api}/sendMessage`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          chat_id: this.chatId,
          text,
          parse_mode: 'Markdown',
        }),
      })
    } catch (e) {
      console.error('[Telegram] sendMessage error:', e)
    }
  }

  async startPolling(): Promise<void> {
    if (this.polling) return
    this.polling = true
    console.log('[Telegram] Starting long polling...')

    const poll = async () => {
      while (this.polling) {
        const updates = await this.getUpdates()
        for (const update of updates) {
          this.offset = update.update_id + 1
          if (update.message && update.message.chat) {
            this.chatId = update.message.chat.id.toString()
            const msg: PlatformMessage = {
              id: update.update_id.toString(),
              text: update.message.text || '',
              from: update.message.from?.username || 'unknown',
              timestamp: update.message.date * 1000,
              platform: 'telegram',
              chatId: this.chatId,
            }
            this.handlers.forEach(h => h(msg))
          }
        }
        await new Promise(r => setTimeout(r, 500))  // 500ms delay between polls
      }
    }

    poll()
  }

  stopPolling(): void {
    this.polling = false
    if (this.pollInterval) {
      clearInterval(this.pollInterval)
      this.pollInterval = null
    }
  }

  onMessage(handler: (msg: PlatformMessage) => void): void {
    this.handlers.push(handler)
  }

  removeHandler(handler: (msg: PlatformMessage) => void): void {
    this.handlers = this.handlers.filter(h => h !== handler)
  }

  private chatId = 'telegram-user'
}
```

### 3.2 OAuth2 Token管理

```typescript
// src/platform/FeishuAdapter.ts 完整实现

import type { PlatformAdapter, PlatformMessage } from './PlatformAdapter'

const FEISHU_API = 'https://open.feishu.cn/open-apis'

interface FeishuToken {
  access_token: string
  expires_in: number
  refresh_token?: string
}

export class FeishuAdapter implements PlatformAdapter {
  platform = 'feishu' as const
  private appId: string
  private appSecret: string
  private accessToken: string | null = null
  private tokenExpiry = 0
  private handlers: ((msg: PlatformMessage) => void)[] = []
  private chatId = 'feishu-user'

  constructor() {
    this.appId = import.meta.env.VITE_FEISHU_APP_ID || ''
    this.appSecret = import.meta.env.VITE_FEISHU_APP_SECRET || ''
  }

  private async getAccessToken(): Promise<string | null> {
    if (this.accessToken && Date.now() < this.tokenExpiry) {
      return this.accessToken
    }

    if (!this.appId || !this.appSecret) {
      console.log('[Feishu] App credentials not configured')
      return null
    }

    try {
      const resp = await fetch(`${FEISHU_API}/auth/v3/tenant_access_token/internal`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ app_id: this.appId, app_secret: this.appSecret }),
      })
      const data = await resp.json()
      if (data.code === 0) {
        this.accessToken = data.tenant_access_token
        this.tokenExpiry = Date.now() + (data.expire - 60) * 1000  // 提前1分钟过期
        return this.accessToken
      }
    } catch (e) {
      console.error('[Feishu] getAccessToken error:', e)
    }
    return null
  }

  async sendMessage(text: string): Promise<void> {
    const token = await this.getAccessToken()
    if (!token) return

    try {
      await fetch(`${FEISHU_API}/im/v1/messages?receive_id_type=open_id`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${token}`,
        },
        body: JSON.stringify({
          receive_id: this.chatId,
          msg_type: 'text',
          content: JSON.stringify({ text }),
        }),
      })
    } catch (e) {
      console.error('[Feishu] sendMessage error:', e)
    }
  }

  // Feishu使用Webhook接收消息（需要公网URL）
  // 本地开发时使用轮询模式
  async startPolling(): Promise<void> {
    console.log('[Feishu] Polling mode - requires webhook URL for production')
  }

  onMessage(handler: (msg: PlatformMessage) => void): void {
    this.handlers.push(handler)
  }

  removeHandler(handler: (msg: PlatformMessage) => void): void {
    this.handlers = this.handlers.filter(h => h !== handler)
  }
}
```
