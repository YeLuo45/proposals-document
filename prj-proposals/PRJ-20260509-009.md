## 1. 背景与目标

当前Agent需要用户显式发指令才行动。但用户行为本身蕴含上下文信号：
- 打开App的时间（深夜？工作日早晨？）
- 连续操作失败（可能需要帮助）
- 长时间无操作（可能离开了）
- 特定页面停留（有兴趣）

**目标：** 基于场景信号自动推断用户需求，主动提供合适的服务。

## 2. 场景信号定义

### 2.1 时间场景
```typescript
// src/services/scene/timeScene.ts

export type TimeScene = 
  | 'morning'      // 6:00-9:00
  | 'work'         // 9:00-12:00
  | 'afternoon'    // 12:00-14:00
  | 'late_afternoon' // 14:00-18:00
  | 'evening'      // 18:00-22:00
  | 'night'        // 22:00-6:00

export function getTimeScene(): TimeScene {
  const hour = new Date().getHours()
  if (hour >= 6 && hour < 9) return 'morning'
  if (hour >= 9 && hour < 12) return 'work'
  if (hour >= 12 && hour < 14) return 'afternoon'
  if (hour >= 14 && hour < 18) return 'late_afternoon'
  if (hour >= 18 && hour < 22) return 'evening'
  return 'night'
}

export function isWeekend(): boolean {
  const day = new Date().getDay()
  return day === 0 || day === 6
}
```

### 2.2 用户状态场景
```typescript
// src/services/scene/userStateScene.ts

export type UserState = 
  | 'active'        // 正在操作
  | 'idle'          // 闲置（5分钟无操作）
  | 'returning'     // 重新回来（闲置后恢复操作）
  | 'struggling'    // 操作失败2次以上
  | 'deep_focus'    // 长时间专注某页面

export interface UserStateSignal {
  state: UserState
  timestamp: number
  context: {
    lastAction?: string
    errorCount?: number
    focusTarget?: string
    idleDuration?: number
  }
}
```

### 2.3 组合场景
```typescript
// 时间 + 用户状态 = 组合场景
type CompositeScene = `${TimeScene}_${UserState}`

// 例如：
// 'morning_active'  — 早晨活跃工作
// 'night_idle'      — 深夜闲置
// 'evening_struggling' — 晚上遇到困难
```

## 3. 场景响应规则

### 3.1 场景响应映射
```typescript
// src/services/scene/sceneResponseMap.ts

import type { CompositeScene } from './types'

interface SceneResponse {
  scene: CompositeScene | TimeScene | UserState
  triggerCondition: (context: SceneContext) => boolean
  action: {
    type: 'suggest' | 'offer_help' | 'remind' | 'adapt' | 'none'
    message: string
    suggestedAction?: string  // 建议的Agent任务
  }
  cooldownMs: number
}

const SCENE_RESPONSES: SceneResponse[] = [
  {
    scene: 'morning_active',
    triggerCondition: (ctx) => ctx.time === 'morning' && ctx.userState === 'active',
    action: { type: 'suggest', message: '早上好！今天有{}项待办，要开始处理吗？', suggestedAction: 'review_tasks' },
    cooldownMs: 3600000,  // 1小时
  },
  {
    scene: 'night_idle',
    triggerCondition: (ctx) => ctx.time === 'night' && ctx.userState === 'idle',
    action: { type: 'suggest', message: '已经很晚了，要不要休息一下？', suggestedAction: 'none' },
    cooldownMs: 1800000,  // 30分钟
  },
  {
    scene: 'evening_struggling',
    triggerCondition: (ctx) => ctx.time === 'evening' && ctx.userState === 'struggling',
    action: { type: 'offer_help', message: '看起来遇到了困难，要我帮忙吗？', suggestedAction: 'analyze_issue' },
    cooldownMs: 300000,  // 5分钟
  },
  {
    scene: 'returning',
    triggerCondition: (ctx) => ctx.userState === 'returning',
    action: { type: 'suggest', message: '欢迎回来！上次你做到了{}，要继续吗？', suggestedAction: 'resume_task' },
    cooldownMs: 600000,  // 10分钟
  },
]
```

## 4. SceneAwarenessEngine

### 4.1 核心类
```typescript
// src/services/scene/SceneAwarenessEngine.ts

import { getTimeScene, isWeekend } from './timeScene'
import type { UserStateSignal } from './userStateScene'
import { SCENE_RESPONSES } from './sceneResponseMap'

class SceneAwarenessEngine {
  private userState: UserStateSignal = { state: 'active', timestamp: Date.now(), context: {} }
  private lastResponseTime = 0
  private idleTimer: ReturnType<typeof setTimeout> | null = null

  // 更新用户状态
  updateUserState(signal: UserStateSignal): void {
    const wasIdle = this.userState.state === 'idle'
    this.userState = signal
    
    if (signal.state === 'returning' && wasIdle) {
      // 从闲置恢复，触发returning场景
    }
    
    // 重置idle计时器
    this.resetIdleTimer()
  }

  // 获取当前场景响应
  getSceneResponse(): SceneResponse | null {
    const now = Date.now()
    const context = this.buildContext()
    
    // 查找匹配的响应规则
    for (const response of SCENE_RESPONSES) {
      if (now - this.lastResponseTime < response.cooldownMs) continue
      if (response.triggerCondition(context)) {
        this.lastResponseTime = now
        return response
      }
    }
    return null
  }

  private buildContext(): SceneContext {
    return {
      time: getTimeScene(),
      isWeekend: isWeekend(),
      userState: this.userState.state,
      ...this.userState.context,
    }
  }

  private resetIdleTimer(): void {
    if (this.idleTimer) clearTimeout(this.idleTimer)
    this.idleTimer = setTimeout(() => {
      this.updateUserState({ state: 'idle', timestamp: Date.now(), context: { idleDuration: 300 } })
    }, 300000)  // 5分钟idle
  }

  // 记录操作（调用此方法来标记用户活跃）
  recordAction(action: string): void {
    this.updateUserState({ state: 'active', timestamp: Date.now(), context: { lastAction: action } })
  }

  // 记录失败
  recordError(): void {
    const errorCount = (this.userState.context.errorCount || 0) + 1
    const state: UserState = errorCount >= 2 ? 'struggling' : 'active'
    this.updateUserState({ state, timestamp: Date.now(), context: { ...this.userState.context, errorCount } })
  }
}

export const sceneAwarenessEngine = new SceneAwarenessEngine()
```

## 5. 与其他模块集成

### 5.1 集成点
```
App.tsx / ChatPanel.tsx
  → recordAction()  用户操作时调用
  → recordError()   操作失败时调用

AgentPanel.tsx
  → sceneAwarenessEngine.getSceneResponse()  定期检查场景响应

memoryManager
  → 场景响应写入记忆，供Agent参考
```

### 5.2 PlatformAdapter集成
场景响应通过 `PlatformAdapter.sendMessage()` 发送给用户。

## 6. 非功能需求

- 场景检测在客户端完成，不上传隐私数据
- 冷却机制防止骚扰用户
- 场景信号本地存储，不上传服务器
- 与现有模块松耦合
