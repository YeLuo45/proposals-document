## 1. 背景

当前多Agent执行结果只能通过console.log查看，用户无法直观看到任务分解和执行状态。

**目标：** 在UI上展示多Agent协作过程的实时状态，让用户看到任务如何被分解、分配和执行。

## 2. 功能设计

### 2.1 MultiAgentStatus组件

```typescript
// src/components/MultiAgent/MultiAgentStatus.tsx

import React, { useState, useEffect } from 'react'

interface TaskStatus {
  id: string
  type: string
  description: string
  assignedTo: string
  status: 'pending' | 'in_progress' | 'completed' | 'failed'
  result?: unknown
  createdAt: number
  completedAt?: number
}

interface MultiAgentStatusProps {
  tasks: TaskStatus[]
  traceId?: string
  onClose?: () => void
}

export const MultiAgentStatus: React.FC<MultiAgentStatusProps> = ({ tasks, traceId, onClose }) => {
  const statusOrder = { pending: 0, in_progress: 1, completed: 2, failed: 3 }

  const sortedTasks = [...tasks].sort((a, b) => {
    const statusDiff = statusOrder[a.status] - statusOrder[b.status]
    if (statusDiff !== 0) return statusDiff
    return a.createdAt - b.createdAt
  })

  const completedCount = tasks.filter(t => t.status === 'completed').length
  const totalCount = tasks.length
  const progress = totalCount > 0 ? (completedCount / totalCount) * 100 : 0

  const getStatusColor = (status: string) => {
    switch (status) {
      case 'pending': return 'var(--color-muted)'
      case 'in_progress': return 'var(--color-warning)'
      case 'completed': return 'var(--color-success)'
      case 'failed': return 'var(--color-error)'
      default: return 'var(--color-muted)'
    }
  }

  const getAgentIcon = (agentId: string) => {
    switch (agentId) {
      case 'orchestrator': return '🎯'
      case 'executor': return '⚡'
      case 'reviewer': return '🔍'
      default: return '🤖'
    }
  }

  return (
    <div className="multi-agent-status">
      <div className="status-header">
        <h3>🤖 多Agent协作</h3>
        {traceId && <span className="trace-id">#{traceId}</span>}
        {onClose && <button onClick={onClose} className="close-btn">×</button>}
      </div>

      <div className="progress-bar">
        <div className="progress-fill" style={{ width: `${progress}%` }} />
        <span className="progress-text">{completedCount}/{totalCount} 完成</span>
      </div>

      <div className="task-list">
        {sortedTasks.map(task => (
          <div key={task.id} className={`task-item ${task.status}`}>
            <div className="task-icon" style={{ color: getStatusColor(task.status) }}>
              {task.status === 'in_progress' ? '⏳' :
               task.status === 'completed' ? '✅' :
               task.status === 'failed' ? '❌' : '⭕'}
            </div>

            <div className="task-info">
              <div className="task-type">{getAgentIcon(task.assignedTo)} {task.type}</div>
              <div className="task-desc">{task.description}</div>
            </div>

            <div className="task-status-badge" style={{ backgroundColor: getStatusColor(task.status) }}>
              {task.status === 'in_progress' ? '执行中' :
               task.status === 'completed' ? '完成' :
               task.status === 'failed' ? '失败' : '等待'}
            </div>
          </div>
        ))}
      </div>
    </div>
  )
}
```

### 2.2 MultiAgentPanel组件（浮动面板）

```typescript
// src/components/MultiAgent/MultiAgentPanel.tsx

import React, { useState, useEffect } from 'react'
import { MultiAgentStatus } from './MultiAgentStatus'

interface TaskRecord {
  id: string
  type: string
  description: string
  assignedTo: string
  status: string
  result?: unknown
  createdAt: number
  completedAt?: number
}

export const MultiAgentPanel: React.FC = () => {
  const [isOpen, setIsOpen] = useState(false)
  const [currentTraceId, setCurrentTraceId] = useState<string | null>(null)
  const [tasks, setTasks] = useState<TaskRecord[]>([])

  // 监听agentBus消息（简化版：轮询console状态）
  useEffect(() => {
    // 实际应该订阅agentBus事件
    // 这里简化处理，每秒检查一次任务状态
    const interval = setInterval(() => {
      // 从Orchestrator获取任务状态（通过console或全局状态）
      const stored = sessionStorage.getItem('multiAgentTasks')
      if (stored) {
        try {
          const parsed = JSON.parse(stored)
          if (parsed.traceId !== currentTraceId) {
            setCurrentTraceId(parsed.traceId)
            setTasks(parsed.tasks || [])
          }
        } catch (e) {
          // ignore
        }
      }
    }, 1000)

    return () => clearInterval(interval)
  }, [currentTraceId])

  if (!isOpen && tasks.length === 0) return null

  return (
    <>
      <button
        className="multi-agent-toggle"
        onClick={() => setIsOpen(!isOpen)}
        title="多Agent协作状态"
      >
        🤖 {tasks.length > 0 && <span className="badge">{tasks.filter(t => t.status === 'in_progress').length}</span>}
      </button>

      {isOpen && (
        <div className="multi-agent-panel">
          <MultiAgentStatus
            tasks={tasks}
            traceId={currentTraceId || undefined}
            onClose={() => setIsOpen(false)}
          />
        </div>
      )}
    </>
  )
}
```
